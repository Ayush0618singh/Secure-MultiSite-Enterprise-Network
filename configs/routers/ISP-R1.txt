enable
configure terminal

hostname ISP-R1

no ip domain-lookup
service password-encryption
enable secret <ENABLE_SECRET>

banner motd # AUTHORIZED NETWORK ADMINISTRATION ONLY #

!
! LOCAL ADMIN ACCOUNT
!

username netadmin privilege 15 secret <LAB_ADMIN_SECRET>

!
! SSH MANAGEMENT
!

ip domain-name enterprise.local
crypto key generate rsa modulus 1024
ip ssh version 2

line console 0
 login local
 exec-timeout 10 0
exit

line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
exit

!
! LINK TO HQ-R1
!

interface GigabitEthernet0/0
 description LINK_TO_HQ-R1
 ip address 203.0.113.1 255.255.255.252
 no shutdown
exit

!
! PUBLIC SERVER NETWORK
!

interface GigabitEthernet0/1
 description PUBLIC_SERVER_NETWORK
 ip address 192.0.2.1 255.255.255.0
 no shutdown
exit

!
! LINK TO BR-R1
!

interface GigabitEthernet0/2
 description LINK_TO_BR-R1
 ip address 198.51.100.1 255.255.255.252
 no shutdown
exit

!
! OSPF DYNAMIC ROUTING
!

router ospf 10
 router-id 2.2.2.2

 network 203.0.113.0 0.0.0.3 area 0
 network 198.51.100.0 0.0.0.3 area 0
 network 192.0.2.0 0.0.0.255 area 0

 passive-interface GigabitEthernet0/1
exit

end
write memory
