enable
configure terminal

hostname HQ-R1

no ip domain-lookup
service password-encryption
enable secret <ENABLE_SECRET>

banner motd # AUTHORIZED NETWORK ADMINISTRATION ONLY #

!
! LOCAL ADMIN ACCOUNT
!

username netadmin privilege 15 secret <LAB_ADMIN_SECRET>

!
! SSH MANAGEMENT
!

ip domain-name enterprise.local
crypto key generate rsa modulus 1024
ip ssh version 2

line console 0
 login local
 exec-timeout 10 0
exit

line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
exit

!
! PHYSICAL LAN INTERFACE
!

interface GigabitEthernet0/0
 description TRUNK_TO_HQ-SW1
 no ip address
 no shutdown
exit

!
! VLAN 10 - HR
!

interface GigabitEthernet0/0.10
 description HR_GATEWAY
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.192
exit

!
! VLAN 20 - FINANCE
!

interface GigabitEthernet0/0.20
 description FINANCE_GATEWAY

! VLAN 30 - DEVELOPMENT

!

!

!
