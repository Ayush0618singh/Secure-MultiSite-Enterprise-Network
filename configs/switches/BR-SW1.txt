enable
configure terminal

hostname BR-SW1

no ip domain-lookup
service password-encryption
enable secret <ENABLE_SECRET>

banner motd # AUTHORIZED NETWORK ADMINISTRATION ONLY #

!
! VLAN DATABASE
!

vlan 50
 name SALES
exit

vlan 60
 name BRANCH_IT
exit

vlan 99
 name MANAGEMENT
exit

vlan 999
 name UNUSED_NATIVE
exit

!
! ROUTER TRUNK TO BR-R1
!

interface FastEthernet0/1
 description TRUNK_TO_BR-R1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 50,60,99,999
 no shutdown
exit

!
! SALES USERS
!

interface FastEthernet0/2
 description BR-SALES-PC1
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

interface FastEthernet0/3
 description BR-SALES-PC2
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

!
! BRANCH IT USER
!

interface FastEthernet0/4
 description BR-IT-PC1
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

!
! MANAGEMENT INTERFACE
!

interface Vlan99
 description BR-SW1_MANAGEMENT
 ip address 10.20.99.2 255.255.255.240
 no shutdown
exit

ip default-gateway 10.20.99.1

!
! SPANNING TREE
!

spanning-tree mode pvst
spanning-tree vlan 50,60,99 root primary

!
! UNUSED PORT SECURITY
!

interface range FastEthernet0/5 - 24
 description UNUSED_PORT
 switchport mode access
 switchport access vlan 999
 shutdown
exit

interface range GigabitEthernet0/1 - 2
 description UNUSED_PORT
 switchport mode access
 switchport access vlan 999
 shutdown
exit

!
! LOCAL ADMIN ACCOUNT
!

username netadmin privilege 15 secret <LAB_ADMIN_SECRET>

!
! SSH MANAGEMENT
!

ip domain-name enterprise.local
crypto key generate rsa modulus 1024
ip ssh version 2

line console 0
 login local
 exec-timeout 10 0
exit

line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit

end
write memory
